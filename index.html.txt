let LAST=null;
$('btnCalc').onclick=()=>{
  const s={...DEF,...me().set};
  let work=0,sick=0,per=0,ann=0,abs=0,unp=0,hol=0,holPaid=0,unHr=0;
  DAYS.forEach((x,i)=>{
    if(x.type==='work')work++;
    else if(x.type==='sick')sick++;
    else if(x.type==='personal')per++;
    else if(x.type==='annual')ann++;
    else if(x.type==='absent')abs++;
    else if(x.type==='unpaid')unp++;
    else if(x.type==='holiday'){hol++;if(holidayPaid(i))holPaid++}
    if(TYPES[x.type].pay)unHr+=x.un||0;
  });
  const paidDays=work+sick+per+ann+holPaid;
  const hourRate=s.rate/s.hours;
  const unpaidCut=+(unHr*hourRate).toFixed(2);
  const wage=+(paidDays*s.rate-unpaidCut).toFixed(2);
  const inc=s.inc;
  const gross=+(wage+inc).toFixed(2);
  const base=s.ssoBase==='all'?gross:wage;
  let sso=Math.round(base*s.sso/100);
  if(s.ssoCap>0)sso=Math.min(sso,s.ssoCap);
  const ded=+(sso+s.tax+s.oth).toFixed(2);
  const net=+(gross-ded).toFixed(2);
  const lostAbs=+((abs+(hol-holPaid))*s.rate).toFixed(2);
  LAST={...PINFO,img:IMG,days:JSON.parse(JSON.stringify(DAYS)),set:s,
    st:{work,sick,per,ann,abs,unp,hol,holPaid,paidDays,unHr},
    wage,inc,gross,sso,tax:s.tax,oth:s.oth,othName:s.othName,unpaidCut,ded,net,lostAbs,
